Muove language card: one example for each production of SPEC §9.4.
Each `== production` heading names the grammar rule; the lines under it are a
whole file that compiles. A `# file: NAME` line starts another module.
Templates are declared phrases, not English: a line must match exactly one.

## Rules

== rule
type Door, Hero
static {door: Door} is unlocked
derived {door: Door} opens for {hero: Hero}
rule unlocked-doors-open
  ?door opens for ?hero
  when
    ?door is unlocked
    ?hero is a Hero

== literal
type Hero, Key
static {hero: Hero} holds {key: Key}
static {hero: Hero} is banned
static {hero: Hero} has strength {n: Int}
derived {hero: Hero} is ready with {n: Int}
rule ready
  ?hero is ready with ?total
  when
    ?hero holds ?key
    not ?hero is banned
    ?hero has strength ?s
    ?s > 0
    ?keys = count where ?hero holds ?k
    ?total = ?s + ?keys
    ?bonus in [1, 2]
    ?bonus <= ?total

== aggregate
type Hero, Key
static {hero: Hero} holds {key: Key}
static {key: Key} weighs {n: Int}
derived {hero: Hero} carries {n: Int}
derived {hero: Hero} favours {key: Key}
rule carried
  ?hero carries ?total
  when
    ?hero is a Hero
    ?total = sum ?w where ?hero holds ?k and ?k weighs ?w
rule favourite
  ?hero favours ?best
  when
    ?hero is a Hero
    ?best = first ?k by ?w where ?hero holds ?k and ?k weighs ?w

== conj
type Hero, Key
static {hero: Hero} holds {key: Key}
static {key: Key} is shiny
derived {hero: Hero} has {n: Int} shiny keys
rule shiny
  ?hero has ?n shiny keys
  when
    ?hero is a Hero
    ?n = count where ?hero holds ?k and ?k is shiny

## Requirements

== claim
type Hero, Room
state {hero: Hero} is in {room: Room}
  exactly one room per hero
state {hero: Hero} is armed
Ash is a Hero
Hall is a Room
Vault is a Room
initially
  Ash is in Hall
requirements
  Ash is in Hall
  never Ash is in Vault
  always Ash is in Hall
  Ash is in Vault requires Ash is armed

== quant
type Hero, Room
static {hero: Hero} is in {room: Room}
Ash is a Hero
Hall is a Room
Vault is a Room
Ash is in Hall
requirements
  some ?h: ?h is in Hall
  every ?h: ?h is a Hero
  no ?h where ?h is in Hall: ?h is in Vault

== vars
type Hero, Room
static {hero: Hero} is in {room: Room}
Ash is a Hero
Hall is a Room
Ash is in Hall
requirements
  every ?h, ?r where ?h is in ?r: ?r is a Room

== body
type Hero, Room
static {hero: Hero} is in {room: Room}
static {hero: Hero} is armed
Ash is a Hero
Hall is a Room
Ash is in Hall
Ash is armed
requirements
  Ash is in Hall and Ash is armed

== requirements
type Hero, Room
state {hero: Hero} is in {room: Room}
  exactly one room per hero
Ash is a Hero
Hall is a Room
initially
  Ash is in Hall
requirements for the level
  invariant every ?h: ?h is a Hero
  eventually Ash is in Hall

## Events and handlers

== handler
type Hero, Room
state {hero: Hero} is in {room: Room}
  exactly one room per hero
event move: {hero: Hero} moves to {room: Room}
on ?hero moves to ?there
  when
    ?hero is in ?here
  set ?hero is in ?there
  ensures
    ?hero is in ?there
  may skip

== action
type Hero, Key
state {hero: Hero} holds {key: Key}
state {key: Key} lies here
event pick-up: {hero: Hero} picks up {key: Key}
event drop: {hero: Hero} drops {key: Key}
effect forge: {hero: Hero} forges
event smith: {hero: Hero} smiths
on ?hero picks up ?key
  when
    ?key lies here
  withdraw ?key lies here
  include ?hero holds ?key
on ?hero drops ?key
  when
    ?hero holds ?key
  delete ?key
  emit ?hero forges
on ?hero smiths
  create ?new: Key
  include ?hero holds ?new

## Functions and views

== define
define wrap(index: Int, count: Int): Int
  if index < 0 then count - 1 else if index >= count then 0 else index

  arguments: config, lib, pkgs, ...
  option myConfig.modules.kanata.enable: enable "Kanata keyboard remapping"
  config when config.myConfig.modules.kanata.enable
    hardware.uinput.enable: true

== args

## Expressions

== expr
nix file
  arguments: pkgs
  let xs = [1, 2]
  let r = {a: 1}
  let f = (x) => x + 1
  value: if true then (with pkgs: hello) else f(r.a)

== value-literal
nix file
  let a = 1
  let b = "text {a}"
  let c = '''
    raw {{b}}
    '''
  let d = true
  let e = false
  value: null

== binop
nix file
  let a = 1 + 2 - 3 * 4
  let b = [1] ++ [2]
  let c = {a: 1} // {b: 2}
  value: true and a == 0 or false

== cmp

  value: [a == 1, a != 2, a < 3, a <= 4, a > 0, a >= 1]
